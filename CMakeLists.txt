cmake_minimum_required(VERSION 3.20)
project(OpenUVPacker VERSION 0.1.0 LANGUAGES CXX)
find_package(Threads REQUIRED)
add_library(OpenUVCore STATIC engine/src/core.cpp)
target_include_directories(OpenUVCore PUBLIC engine/include)
target_compile_features(OpenUVCore PUBLIC cxx_std_20)
add_executable(openuv_engine engine/src/main.cpp)
target_compile_features(openuv_engine PRIVATE cxx_std_20)
target_link_libraries(openuv_engine PRIVATE OpenUVCore Threads::Threads)
enable_testing()
add_executable(openuv_tests tests/native_tests.cpp)
target_link_libraries(openuv_tests PRIVATE OpenUVCore)
add_test(NAME core_geometry COMMAND openuv_tests)
add_test(NAME native_selftest COMMAND openuv_engine --self-test)
install(TARGETS openuv_engine RUNTIME DESTINATION addon/bin)
